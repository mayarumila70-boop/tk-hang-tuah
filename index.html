 * GOOGLE APPS SCRIPT BACKEND (appscript.gs)
 * Media Pembelajaran PAUD "Aku Mengenal Tanaman"

// Handle GET Requests (Ambil data dari Sheet)
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === "getMateri") {
    var sheet = ss.getSheetByName("Materi");
    var data = sheet.getDataRange().getValues();
    return createJsonResponse({ status: "success", data: data });
  } 
  else if (action === "getVideo") {
    var sheet = ss.getSheetByName("Video");
    var data = sheet.getDataRange().getValues();
    return createJsonResponse({ status: "success", data: data });
  } 
  else if (action === "getHasil") {
    var sheet = ss.getSheetByName("Hasil");
    var data = sheet.getDataRange().getValues();
    return createJsonResponse({ status: "success", data: data });
  }
  
  return createJsonResponse({ status: "error", message: "Action tidak dikenal" });
}

// Handle POST Requests (Simpan hasil kuis & Buat Sertifikat)
function doPost(e) {
  try {
    var postData = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetHasil = ss.getSheetByName("Hasil");
    
    var nama = postData.nama;
    var tanggal = new Date().toLocaleDateString("id-ID");
    var skor = postData.skor;
    var status = skor >= 80 ? "LULUS (Sertifikat)" : "BBM (Belum Belajar Maksimal)";
    
    // Simpan ke Google Spreadsheet
    sheetHasil.appendRow([nama, tanggal, skor, status]);
    
    // Buat Sertifikat Digital PDF jika LULUS
    var certUrl = "";
    if (skor >= 80) {
      certUrl = generateDigitalCertificate(nama, skor, tanggal);
    }
    
    return createJsonResponse({
      status: "success",
      message: "Data berhasil disimpan!",
      certUrl: certUrl
    });
  } catch (error) {
    return createJsonResponse({ status: "error", message: error.toString() });
  }
}

// Helper Response JSON
function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// Fungsi Membuat Sertifikat PDF Otomatis via Google Docs Template
function generateDigitalCertificate(nama, skor, tanggal) {
  // Ganti ID Template Google Docs dan ID Folder Penyimpanan PDF berikut:
  var TEMPLATE_DOC_ID = "MASUKKAN_ID_GOOGLE_DOC_TEMPLATE_ANDA";
  var TARGET_FOLDER_ID = "MASUKKAN_ID_FOLDER_GOOGLE_DRIVE_ANDA";
  
  try {
    var templateFile = DriveApp.getFileById(TEMPLATE_DOC_ID);
    var targetFolder = DriveApp.getFolderById(TARGET_FOLDER_ID);
    
    var copyDoc = templateFile.makeCopy("Sertifikat_" + nama, targetFolder);
    var doc = DocumentApp.openById(copyDoc.getId());
    var body = doc.getBody();
    
    // Replace Tags Placeholder
    body.replaceText("{{NAMA_ANAK}}", nama);
    body.replaceText("{{SKOR}}", skor.toString());
    body.replaceText("{{TANGGAL}}", tanggal);
    doc.saveAndClose();
    
    // Convert to PDF
    var pdfBlob = copyDoc.getAs("application/pdf");
    var pdfFile = targetFolder.createFile(pdfBlob);
    pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    // Hapus temporary doc file
    DriveApp.getFileById(copyDoc.getId()).setTrashed(true);
    
    return pdfFile.getUrl();
  } catch(e) {
    return "Gagal membuat PDF: " + e.toString();
  }
}

Struktur Google Spreadsheet Database

Buat Google Spreadsheet baru lalu buat 4 Sheet berikut dengan header kolom persis di bawah ini:

  • Sheet 1: Materi (Kolom A: ID, Kolom B: Judul, Kolom C: Isi Materi, Kolom D: URL Gambar, Kolom E: URL Audio)
  • Sheet 2: Video (Kolom A: Judul Video, Kolom B: URL Video YouTube)
  • Sheet 3: Soal (Kolom A: Nomor, Kolom B: Pertanyaan, Kolom C: Pilihan A, Kolom D: Pilihan B, Kolom E: Pilihan C, Kolom F: Jawaban Benar)
  • Sheet 4: Hasil (Kolom A: Nama Anak, Kolom B: Tanggal, Kolom C: Skor, Kolom D: Status)

Template Google Docs Sertifikat

Buat dokumen Google Docs baru sebagai template sertifikat anak. Gunakan tag berikut di dalam dokumen:

Diberikan kepada: {{NAMA_ANAK}}

Telah menyelesaikan Kuis "Aku Mengenal Tanaman" dengan Nilai: {{SKOR}}

Diberikan pada tanggal: {{TANGGAL}}

Cara Deploy & Hosting Gratis

  1. Buka Google Spreadsheet yang telah dibuat, klik Ekstensi > Apps Script.
  2. Paste seluruh kode appscript.gs di atas. Selesaikan pengisian ID Google Doc & Folder.
  3. Klik Terapkan (Deploy) > Deployment Baru.
  4. Pilih Jenis Deployment: Aplikasi Web (Web App).
  5. Set Akses Siapa Saja (Anyone), lalu klik Terapkan dan berikan izin izin Google Drive.
  6. Salin Web App URL yang dihasilkan dan tempel di **Dashboard Guru** aplikasi ini.
  7. Hosting gratis website ini: Simpan file ini sebagai index.html dan upload ke GitHub Pages, Netlify, atau Vercel!
